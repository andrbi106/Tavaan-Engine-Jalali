## استفاده

```php
use Tavaan\Engine\I18n\Jalali\JalaliConverter;
use Tavaan\Engine\I18n\Jalali\JalaliDate;

// تبدیل میلادی به جلالی
[$jy, $jm, $jd] = JalaliConverter::gregorianToJalali(2024, 3, 20);
// [1403, 1, 1]

// تبدیل جلالی به میلادی
[$gy, $gm, $gd] = JalaliConverter::jalaliToGregorian(1403, 1, 1);
// [2024, 3, 20]

// سال کبیسه
JalaliConverter::isLeapJalali(1403);  // true
JalaliConverter::isLeapJalali(1402);  // false

// طول ماه
JalaliConverter::jalaliMonthLength(1403, 12);  // 30 (leap year)
JalaliConverter::jalaliMonthLength(1402, 12);  // 29 (non-leap)

// روز سال
JalaliConverter::dayOfYear(1403, 1, 1);   // 1
JalaliConverter::dayOfYear(1403, 12, 30); // 366 (leap)

// فرمت
JalaliDate::format(1403, 1, 15, 'Y/m/d');     // '1403/01/15'
JalaliDate::format(1403, 1, 15, 'Y M d');     // '1403 فروردین 15'
JalaliDate::format(1403, 1, 15, 'y/n/j');     // '03/1/15'

// نام ماه و روز
JalaliDate::monthName(1);                     // 'فروردین'
JalaliDate::seasonName(1);                    // 'بهار'
JalaliDate::weekday(1403, 1, 1);              // 'چهارشنبه'

// امروز
[$y, $m, $d] = JalaliDate::today();
```
